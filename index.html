-- Automated Mining Cadastre Overlap Detection Query
SELECT
  a.claim_id AS applicant_claim,
  b.claim_id AS existing_claim,
  ST_Area(ST_Intersection(a.geom, b.geom)) / 10000.0 AS overlap_hectares,
  ST_Overlaps(a.geom, b.geom) AS has_spatial_conflict
FROM mining_claims a
JOIN mining_claims b ON ST_Intersects(a.geom, b.geom)
WHERE a.claim_id != b.claim_id AND a.status = 'PENDING';
